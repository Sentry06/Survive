local WeaponRegistry = require(
	game.ServerScriptService.Weapons:WaitForChild("WeaponRegistry")
)

local RewardGenerator = {}

-- =========================
-- POSSIBLE ATTRIBUTES
-- =========================
local possibleAttributes = {
	"Damage",
	"Cooldown",
	"Range",
	"MultiAttack",
	"MovementSpeed",
	"AttackSize",
	"ProjectileSpeed",
	"Luck",
	"Shield",
	"ShieldRegenDelay",
	"MaxHP",
	"LifeRegen",
	"ProjectilePierce",
	"AreaDamage",
}

-- =========================
-- GENERATE REWARDS
-- =========================
function RewardGenerator:Generate(player, rarity, chestDef)

	local rewards = {
		Gold = 0,
		Attributes = {},
		Weapon = nil,
		WeaponImage = nil
	}

	-- 🪙 GOLD
	rewards.Gold = math.random(chestDef.GoldMin, chestDef.GoldMax)

	-- 📈 ATTRIBUTES
	local used = {}

	for i = 1, chestDef.AttributeCount do

		local attributeName

		repeat
			attributeName = possibleAttributes[
			math.random(1, #possibleAttributes)
			]
		until not used[attributeName]

		used[attributeName] = true

		local attrData = {
			Name = attributeName
		}

		-- atributos multiplicativos
		local MULTIPLIER_ATTRS = {
			Damage = true,
			Range = true,
			MaxHP = true,
			MovementSpeed = true,
			AttackSize = true,
			ProjectileSpeed = true,
			Cooldown = true,
			ShieldRegenDelay = true,
		}

		-- atributos absolutos
		local ABSOLUTE_ATTRS = {
			MultiAttack = true,
			ProjectilePierce = true,
			Shield = true,
			LifeRegen = true,
			Luck = true,
			AreaDamage = true,
		}

		if MULTIPLIER_ATTRS[attributeName] then
			attrData.Multiplier = chestDef.UpgradeMultiplier

		elseif ABSOLUTE_ATTRS[attributeName] then
			attrData.Value = 1
		end

		table.insert(rewards.Attributes, attrData)
	end


	-- ⚔ WEAPON CHANCE
	-- ⚔ WEAPON CHANCE
	if math.random() <= chestDef.WeaponChance then

		local possibleWeapons = WeaponRegistry:GetAll()
		local weaponList = {}

		for id, data in pairs(possibleWeapons) do
			table.insert(weaponList, {
				Id = data.Id,
				Image = data.Image
			})
		end

		if #weaponList > 0 then
			local chosen = weaponList[math.random(1, #weaponList)]

			rewards.Weapon = chosen.Id
			rewards.WeaponImage = chosen.Image

			print("🎯 Arma gerada no baú:", chosen.Id)
		end
	end

	return rewards
end

return RewardGenerator
