-- BossSpawner.server
-- ETAPA 2.1 — VERSÃO CORRIGIDA (XP + LEVEL UP CONSISTENTE)

local RunService = game:GetService("RunService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Players = game:GetService("Players")
local Debris = game:GetService("Debris")

local XPService = require(game.ServerScriptService.XPService)
local DifficultyService = require(game.ServerScriptService.DifficultyService)
local DamageService = require(game.ServerScriptService.Combat.DamageService)
local GoldService = require(game.ServerScriptService.GoldService)
local EnemyTimeController = require(
	game.ServerScriptService.Combat:WaitForChild("EnemyTimeController")
)
local ArtifactRegistry = require(
	game.ServerScriptService.Artifacts.ArtifactRegistry
)
local EnemyTimeController = require(
	game.ServerScriptService.Combat.EnemyTimeController
)

local ChestSpawner = require(game.ServerScriptService.Chests.ChestSpawner)

local function tryDropArtifact(position, chance)
	if math.random() > chance then
		return
	end

	local artifacts = ArtifactRegistry:GetAll()
	if not artifacts then return end

	-- coleta IDs REAIS (strings)
	local ids = {}

	for key, value in pairs(artifacts) do
		if typeof(key) == "string" then
			table.insert(ids, key)
		elseif typeof(value) == "table" and typeof(value.Id) == "string" then
			table.insert(ids, value.Id)
		end
	end

	if #ids == 0 then
		warn("❌ Nenhum artefato válido no Registry")
		return
	end

	local artifactId = ids[math.random(1, #ids)]

	-- pasta no workspace
	local artifactsFolder = workspace:FindFirstChild("Artifacts")
	if not artifactsFolder then
		artifactsFolder = Instance.new("Folder")
		artifactsFolder.Name = "Artifacts"
		artifactsFolder.Parent = workspace
	end
	
	local ArtifactDropService = require(
		game.ServerScriptService.Artifacts.ArtifactDropService
	)

	ArtifactDropService:Spawn(artifactId, position)
end

-- =========================
-- GAME STATE
-- =========================
local GameState = ReplicatedStorage:WaitForChild("GameState")
local GamePaused = GameState:WaitForChild("GamePaused")

_G.Enemies = _G.Enemies or {}

-- =========================
-- CONFIG
-- =========================
local BOSS_INTERVAL = 10
local BASE_BOSS_HP = 180
local BASE_BOSS_XP = 120
local BOSS_SPEED = 11

local ATTACK_RANGE = 8
local ATTACK_COOLDOWN = 1.5

-- =========================
-- STATE
-- =========================
local bossAlive = false
local bossTimer = 0

-- =========================
-- FOLDER
-- =========================
local enemiesFolder = workspace:FindFirstChild("Enemies")
if not enemiesFolder then
	enemiesFolder = Instance.new("Folder")
	enemiesFolder.Name = "Enemies"
	enemiesFolder.Parent = workspace
end

-- =========================
-- UTIL
-- =========================
local function getPlayerCenter()
	for _, player in ipairs(Players:GetPlayers()) do
		local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
		if root then
			return root.Position
		end
	end
	return Vector3.new(0, 5, 0)
end

local function getBossSpawnPos()
	return getPlayerCenter() + Vector3.new(0, 0, -60)
end

-- =========================
-- BOSS AI
-- =========================
local function startBossAI(bossData, speed)
	local humanoid = bossData.Humanoid
	local root = bossData.Root
	local lastAttack = 0

	bossData.AIConn = RunService.Heartbeat:Connect(function(dt)
		if not bossData.Alive or humanoid.Health <= 0 then
			return
		end

		if GamePaused.Value or EnemyTimeController:IsFrozen() then
			humanoid.WalkSpeed = 0
			humanoid:MoveTo(root.Position)
			return
		end

		local closestPlayer
		local closestRoot
		local minDist = math.huge

		for _, player in ipairs(Players:GetPlayers()) do
			local pr = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
			if pr then
				local d = (pr.Position - root.Position).Magnitude
				if d < minDist then
					minDist = d
					closestPlayer = player
					closestRoot = pr
				end
			end
		end

		if closestRoot then
			local slow = EnemyTimeController:GetSlow()
			humanoid.WalkSpeed = speed * slow
			humanoid:MoveTo(closestRoot.Position)

			lastAttack += dt
			if minDist <= ATTACK_RANGE and lastAttack >= ATTACK_COOLDOWN then
				local diff = DifficultyService:GetStats() or {}
				local damageMultiplier = diff.DamageMultiplier
				if typeof(damageMultiplier) ~= "number" then
					damageMultiplier = 1
				end

				DamageService:TryDamagePlayer(
					root, -- HumanoidRootPart do BOSS
					20 * damageMultiplier,
					8 -- alcance maior que inimigos comuns
				)

				lastAttack = 0
			end
		end
	end)
end

-- =========================
-- SPAWN BOSS
-- =========================
local function spawnBoss()
	if bossAlive or not _G.GameStarted or GamePaused.Value then return end

	local bossTemplate = ReplicatedStorage.Enemies:FindFirstChild("BossEnemy")
	if not bossTemplate then
		warn("❌ BossEnemy não encontrado")
		return
	end

	local diff = DifficultyService:GetStats()

	local boss = bossTemplate:Clone()
	local humanoid = boss:FindFirstChildOfClass("Humanoid")
	local root = boss:FindFirstChild("HumanoidRootPart")
	if not humanoid or not root then
		boss:Destroy()
		return
	end

	local maxHP = math.floor(BASE_BOSS_HP * diff.HealthMultiplier)
	humanoid.MaxHealth = maxHP
	humanoid.Health = maxHP
	humanoid.WalkSpeed = BOSS_SPEED

	root.CFrame = CFrame.new(getBossSpawnPos())
	boss.Parent = enemiesFolder

	local bossData = {
		Alive = true,
		Humanoid = humanoid,
		Root = root,
		Model = boss,
		IsBoss = true
	}

	table.insert(_G.Enemies, bossData)

	bossAlive = true
	bossTimer = 0

	startBossAI(bossData, BOSS_SPEED)

	print("👑 BOSS SPAWNADO")

	-- =========================
	-- MORTE DO BOSS
	-- =========================
	humanoid.Died:Connect(function()
		bossData.Alive = false
		bossAlive = false

		if bossData.AIConn then
			bossData.AIConn:Disconnect()
		end

		for i, e in ipairs(_G.Enemies) do
			if e == bossData then
				table.remove(_G.Enemies, i)
				break
			end
		end

		-- 🔥 XP + LEVEL UP (MESMO FLUXO DO EnemySpawner)
		local closestPlayer
		local minDist = math.huge

		for _, p in ipairs(Players:GetPlayers()) do
			local pr = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
			if pr then
				local d = (pr.Position - root.Position).Magnitude
				if d < minDist then
					minDist = d
					closestPlayer = p
				end
			end
		end

		if closestPlayer then
			local ms = closestPlayer:FindFirstChild("MatchStats")
			if ms then
				local xp = ms.XP
				local level = ms.Level
				local xpToNext = ms.XPToNext

				xp.Value += BASE_BOSS_XP
				print("👑 BOSS DERROTADO → XP bônus:", BASE_BOSS_XP)

				while xp.Value >= xpToNext.Value do
					xp.Value -= xpToNext.Value
					level.Value += 1
					xpToNext.Value = XPService:GetXPRequired(level.Value)
				end
			end
			GoldService:GiveRunGold(closestPlayer, 25, "Boss")
		end
		
		-- 🧿 DROP DE ARTEFATO (BOSS)
		local dropPosition = root.Position
		tryDropArtifact(dropPosition, 1) -- 100% de chance

		if closestPlayer then
			local randomOffset = Vector3.new(
				math.random(-6,6),
				0,
				math.random(-6,6)
			)

			ChestSpawner:SpawnChest(dropPosition + randomOffset, closestPlayer)
		end

		Debris:AddItem(boss, 5)
	end)
end

-- =========================
-- LOOP
-- =========================
RunService.Heartbeat:Connect(function(dt)
	if not _G.GameStarted or GamePaused.Value then return end

	bossTimer += dt
	if bossTimer >= BOSS_INTERVAL then
		spawnBoss()
	end
end)

print("👑 BossSpawner CORRIGIDO (XP + LevelUp consistente)")
