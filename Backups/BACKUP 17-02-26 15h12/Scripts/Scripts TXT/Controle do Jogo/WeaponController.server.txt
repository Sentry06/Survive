-- WeaponController.server
-- Versão revisada e blindada

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local WeaponController = {}

local GameState = ReplicatedStorage:WaitForChild("GameState")
local GamePaused = GameState:WaitForChild("GamePaused")

local WeaponsFolder = game.ServerScriptService:WaitForChild("Weapons")

local WeaponSlots = require(WeaponsFolder:WaitForChild("WeaponSlots"))
local ProjectileLimiter = require(WeaponsFolder:WaitForChild("ProjectileLimiter"))

local DEFAULT_SLOTS = 5
local MAX_PROJECTILES_PER_PLAYER = 30

local playerSlots = {}
_G.PlayerWeaponSlots = playerSlots

-- =========================
-- AUX
-- =========================

local function getStartingWeapon(player)
	local stats = player:FindFirstChild("GlobalStats")
	if stats and stats:FindFirstChild("SelectedWeapon") then
		return stats.SelectedWeapon.Value
	end
	return "Knife"
end

local function getMaxSlots(player)
	local stats = player:FindFirstChild("GlobalStats")
	if stats and stats:FindFirstChild("ExtraWeaponSlots") then
		return DEFAULT_SLOTS + stats.ExtraWeaponSlots.Value
	end
	return DEFAULT_SLOTS
end

-- =========================
-- SETUP
-- =========================

local function setupPlayer(player)

	if playerSlots[player] then
		return
	end

	local maxSlots = getMaxSlots(player)
	local slots = WeaponSlots.new(maxSlots)

	local limiter = ProjectileLimiter.new(MAX_PROJECTILES_PER_PLAYER)
	slots.ProjectileLimiter = limiter

	playerSlots[player] = slots

	local startingWeaponName = getStartingWeapon(player)
	local module = WeaponsFolder:FindFirstChild(startingWeaponName)

	if module then
		local weapon = require(module).new(player, limiter)
		slots:AddWeapon(weapon)
	end
end

Players.PlayerAdded:Connect(function(player)
	player.CharacterAdded:Connect(function()
		setupPlayer(player)
	end)
end)

Players.PlayerRemoving:Connect(function(player)
	playerSlots[player] = nil
end)

-- =========================
-- LOOP
-- =========================
RunService.Heartbeat:Connect(function(dt)

	if GamePaused.Value then return end
	if not _G.GameStarted then return end

	for player, slots in pairs(playerSlots) do

		if not player or not player.Parent then
			playerSlots[player] = nil
			continue
		end

		for _, slot in ipairs(slots.Weapons) do
			local weapon = slot.Instance
			if weapon then
				weapon:Tick(dt)
				weapon:TryAttack()
			end
		end
	end
end)

-- =========================
-- GIVE / UPGRADE
-- =========================

function WeaponController:GiveWeapon(player, weaponName, rarity)

	if not weaponName then return false end

	local module = WeaponsFolder:FindFirstChild(weaponName)
	if not module then
		warn("Weapon not found:", weaponName)
		return false
	end

	local slots = playerSlots[player]
	if not slots then return false end

	local rarityLevels = {
		Bronze = 1,
		Silver = 2,
		Gold = 3,
		Diamond = 5
	}

	local levelIncrease = rarityLevels[rarity] or 1

	-- Se já tem a arma
	local existing = slots:GetWeapon(weaponName)
	if existing then

		if existing and existing.Instance and type(existing.Instance.OnUpgrade) == "function" then

			local currentLevel = existing.Level or 1
			local newLevel = currentLevel + levelIncrease

			existing.Level = newLevel
			existing.Instance:OnUpgrade(newLevel)
		end

		return true
	end

	-- Slot livre
	if slots:HasFreeSlot() then

		local weapon = require(module).new(player, slots.ProjectileLimiter)
		local success = slots:AddWeapon(weapon)

		if success then
			print("⚔ Weapon added:", weaponName)
			return true
		end

		return false
	end

	-- Slots cheios → evolui aleatória
	if #slots.Weapons > 0 then

		local randomWeapon =
			slots.Weapons[math.random(1, #slots.Weapons)]

		if type(randomWeapon.OnUpgrade) == "function" then
			local currentLevel = randomWeapon.Level or 1
			local newLevel = currentLevel + levelIncrease
			randomWeapon:OnUpgrade(newLevel)

			print("🔥 Slots cheios → evoluiu:", randomWeapon.Name)
			return true
		end
	end

	return false
end

print("⚔️ WeaponController FINAL revisado")
_G.WeaponController = WeaponController
return WeaponController
