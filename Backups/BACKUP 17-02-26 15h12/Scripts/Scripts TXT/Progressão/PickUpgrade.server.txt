-- PickUpgrade.server
-- APLICA ATRIBUTOS (SEM CONTROLE DE UI)

local ReplicatedStorage = game:GetService("ReplicatedStorage")

local Remotes = ReplicatedStorage:WaitForChild("Remotes")
local PickUpgrade = Remotes:WaitForChild("PickUpgrade")

-- 📈 atributos multiplicativos
local MULTIPLIER_ATTRS = {
	Damage = true,
	Range = true,
	MaxHP = true,
}

-- ⏱️ atributos onde MENOR é melhor
local INVERT_MULTIPLIER = {
	Cooldown = true,
	ShieldRegenDelay = true,
}

PickUpgrade.OnServerEvent:Connect(function(player, data)
	if data.Action ~= "PICK" then return end
	if data.Type ~= "Attribute" then return end

	local attrs = player:FindFirstChild("MatchStats")
		and player.MatchStats:FindFirstChild("Attributes")
	if not attrs then return end

	local stat = attrs:FindFirstChild(data.Id)
	if not stat then
		warn("⚠️ Upgrade inválido recebido:", data.Id)
		return
	end

	-- =========================
	-- MULTI ATTACK (SEM LIMITE)
	-- =========================
	if data.Id == "MultiAttack" then
		local gainByRarity = {
			Bronze = 1,
			Silver = 2,
			Gold = 3,
			Diamond = 5,
		}

		stat.Value += gainByRarity[data.Rarity] or 1

		-- =========================
		-- SHIELD (ABSOLUTO)
		-- =========================
	elseif data.Id == "Shield" then
		local gainByRarity = {
			Bronze = 20,
			Silver = 30,
			Gold = 40,
			Diamond = 70,
		}

		local gain = gainByRarity[data.Rarity] or 20
		stat.Value += gain

		local current = attrs:FindFirstChild("CurrentShield")
		if current then
			current.Value = math.min(current.Value + gain, stat.Value)
		end

		-- =========================
		-- LIFE REGEN (ABSOLUTO)
		-- =========================
	elseif data.Id == "LifeRegen" then
		local gainByRarity = {
			Bronze = 0.5,
			Silver = 1.0,
			Gold = 2.0,
			Diamond = 5.0,
		}

		stat.Value += gainByRarity[data.Rarity] or 0.5

		-- =========================
		-- PROJECTILE PIERCE
		-- =========================
	elseif data.Id == "ProjectilePierce" then
		local gainByRarity = {
			Bronze = 1,
			Silver = 2,
			Gold = 3,
			Diamond = 5,
		}

		stat.Value += gainByRarity[data.Rarity] or 1

		-- =========================
		-- AREA DAMAGE (PERCENTUAL)
		-- =========================
	elseif data.Id == "AreaDamage" then
		stat.Value += data.Value

		player:SetAttribute("AreaDamage", stat.Value)

		-- =========================
		-- MULTIPLICADORES INVERTIDOS
		-- =========================
	elseif INVERT_MULTIPLIER[data.Id] then
		stat.Value *= (1 - data.Value)

		if data.Id == "Cooldown" and stat.Value < 0.1 then
			stat.Value = 0.1
		end

		-- =========================
		-- MULTIPLICADORES NORMAIS
		-- =========================
	elseif MULTIPLIER_ATTRS[data.Id] then
		stat.Value *= (1 + data.Value)

		-- =========================
		-- ADITIVO PADRÃO
		-- =========================
	else
		stat.Value += data.Value
	end

	-- =========================
	-- LOG
	-- =========================
	local valueStr =
		(data.Id == "MultiAttack" or data.Id == "Shield" or data.Id == "ProjectilePierce")
		and tostring(stat.Value)
		or string.format("%.2f", stat.Value)

	print(
		string.format(
			"⬆️ %s (%s) aplicado → %s",
			data.Id,
			data.Rarity or "?",
			valueStr
		)
	)
end)

print("📩 PickUpgrade FINAL (ESTÁVEL)")
