function WeaponRegistry:GetRandomWeaponId(player)
	local list = {}

	for id, _ in pairs(self.Definitions) do
		table.insert(list, id)
	end

	return list[math.random(1, #list)]
end

function WeaponRegistry:GetAll()
	return self.Definitions
end
