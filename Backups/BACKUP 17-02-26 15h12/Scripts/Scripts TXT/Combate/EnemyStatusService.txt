local RunService = game:GetService("RunService")

local EnemyStatusService = {}

local frozenEnemies = {}
local electrifiedEnemies = {}

local Debris = game:GetService("Debris")

local function createLightningBeam(pos1, pos2)

	local direction = pos2 - pos1
	local distance = direction.Magnitude

	local beam = Instance.new("Part")
	beam.Anchored = true
	beam.CanCollide = false
	beam.Material = Enum.Material.Neon
	beam.Color = Color3.fromRGB(0,170,255)
	beam.Size = Vector3.new(0.3, 0.3, distance)
	beam.CFrame =
		CFrame.new(pos1, pos2) *
		CFrame.new(0, 0, -distance / 2)

	beam.Parent = workspace

	Debris:AddItem(beam, 0.12)
end

-------------------------------------------------
-- ❄ FREEZE (continua igual)
-------------------------------------------------
function EnemyStatusService:Freeze(enemyData, duration)

	if not enemyData or not enemyData.Humanoid then
		return
	end

	if frozenEnemies[enemyData] then
		return
	end

	local humanoid = enemyData.Humanoid
	local originalSpeed = humanoid.WalkSpeed

	frozenEnemies[enemyData] = {
		OriginalSpeed = originalSpeed
	}

	humanoid.WalkSpeed = 0

	task.delay(duration, function()
		if humanoid and humanoid.Parent then
			humanoid.WalkSpeed = originalSpeed
		end
		frozenEnemies[enemyData] = nil
	end)
end

-------------------------------------------------
-- ⚡ ELECTRIFY
-------------------------------------------------
function EnemyStatusService:Electrify(enemyData, sourcePlayer, damage, duration, tick, radius)

	if not enemyData or not enemyData.Humanoid then
		return
	end

	if electrifiedEnemies[enemyData] then
		return -- já eletrificado
	end

	electrifiedEnemies[enemyData] = {
		EndsAt = os.clock() + duration,
		LastTick = 0,
		Damage = damage,
		Tick = tick,
		Radius = radius,
		Source = sourcePlayer
	}

	print("⚡ Inimigo eletrificado")
end

function EnemyStatusService:ChainLightning(startEnemy, player, baseDamage)

	if not startEnemy or not startEnemy.Humanoid then return end

	local attrs = player:GetAttributes()

	local pierce = math.floor(attrs.ProjectilePierce or 0)
	local areaPercent = attrs.AreaDamage or 0
	local attackSize = attrs.AttackSize or 1

	local BASE_RADIUS = 8
	local radiusMultiplier = 1 + (areaPercent * 1.5)
	local radius = BASE_RADIUS * attackSize * radiusMultiplier

	local maxJumps = math.max(1, pierce)

	local hitEnemies = {}
	local currentEnemy = startEnemy
	local currentDamage = baseDamage

	for jump = 1, maxJumps do

		if not currentEnemy or not currentEnemy.Humanoid then
			break
		end

		hitEnemies[currentEnemy] = true

		-- ⚡ DANO
		currentEnemy.Humanoid:TakeDamage(currentDamage)

		-- 🔎 Procurar próximo alvo
		local nextEnemy
		local shortestDist = math.huge

		for _, enemy in ipairs(_G.Enemies or {}) do
			if enemy.Alive
				and enemy.Root
				and enemy.Humanoid
				and not hitEnemies[enemy] then

				local dist =
					(enemy.Root.Position - currentEnemy.Root.Position).Magnitude

				if dist <= radius and dist < shortestDist then
					shortestDist = dist
					nextEnemy = enemy
				end
			end
		end

		-- ⚡ Criar raio VISUAL
		if nextEnemy then
			createLightningBeam(
				currentEnemy.Root.Position,
				nextEnemy.Root.Position
			)
		end

		-- 🔻 Decaimento
		currentDamage *= 0.75
		currentEnemy = nextEnemy
	end
end

-------------------------------------------------
-- ⚡ LOOP GLOBAL
-------------------------------------------------
RunService.Heartbeat:Connect(function()

	local now = os.clock()

	for enemyData, info in pairs(electrifiedEnemies) do

		if not enemyData.Alive
			or not enemyData.Humanoid
			or enemyData.Humanoid.Health <= 0 then

			electrifiedEnemies[enemyData] = nil
			continue
		end

		if now >= info.EndsAt then
			electrifiedEnemies[enemyData] = nil
			continue
		end

		if now - info.LastTick >= info.Tick then

			info.LastTick = now

			-- DANO
			enemyData.Humanoid:TakeDamage(info.Damage)

			-- PROPAGAÇÃO
			for _, other in ipairs(_G.Enemies or {}) do
				if other ~= enemyData and other.Alive then

					local dist =
						(other.Root.Position - enemyData.Root.Position).Magnitude

					if dist <= info.Radius then
						if not electrifiedEnemies[other] then
							EnemyStatusService:Electrify(
								other,
								info.Source,
								info.Damage * 0.7, -- redução por salto
								1.5,
								info.Tick,
								info.Radius
							)
						end
					end
				end
			end
		end
	end
end)

return EnemyStatusService
