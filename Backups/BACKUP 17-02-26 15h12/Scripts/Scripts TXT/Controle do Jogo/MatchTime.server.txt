-- MatchTime.server
-- Timer oficial da partida (APENAS GamePaused controla)

local RunService = game:GetService("RunService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Players = game:GetService("Players")

-- Remotes
local Remotes = ReplicatedStorage:WaitForChild("Remotes")
local TimeRemote = Remotes:WaitForChild("MatchTimeUpdate")

-- GameState
local GameState = ReplicatedStorage:WaitForChild("GameState")
local GamePaused = GameState:WaitForChild("GamePaused")

local elapsedTime = 0
local lastSecondSent = -1
local goldGranted = false

local function broadcastTime()
	for _, player in ipairs(Players:GetPlayers()) do
		TimeRemote:FireClient(player, elapsedTime)
	end
end

RunService.Heartbeat:Connect(function(dt)
	-- ⏸️ pausa REAL
	if GamePaused.Value then
		return
	end

	elapsedTime += dt
	-- 🪙 OURO POR SOBREVIVÊNCIA (20 MIN)
	if not goldGranted and elapsedTime >= 1200 then
		goldGranted = true

		for _, player in ipairs(Players:GetPlayers()) do
			local gs = player:FindFirstChild("GlobalStats")
			if gs then
				local gold = gs:FindFirstChild("Gold")
				if gold then
					local amount = 200

					if player:GetAttribute("Artifact_Aurum") then
						amount *= 3
					end

					gold.Value += amount
				end
			end
		end
		GoldService:GiveRunGold(player, 200, "Survival")
		print("🪙 Ouro de sobrevivência concedido (+200)")
	end


	local whole = math.floor(elapsedTime)
	if whole ~= lastSecondSent then
		lastSecondSent = whole
		broadcastTime()
	end
end)

print("⏱️ MatchTime FINAL — independente de GameStarted")
