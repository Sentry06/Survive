-- Chinelo.lua
-- Versão FINAL compatível com WeaponBase atual

local RunService = game:GetService("RunService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Debris = game:GetService("Debris")

local EnemyDamageService = require(
	game.ServerScriptService.Combat.EnemyDamageService
)
local WeaponBase = require(script.Parent:WaitForChild("WeaponBase"))

local Chinelo = {}
Chinelo.__index = Chinelo
setmetatable(Chinelo, { __index = WeaponBase })

local ProjectileTemplate =
	ReplicatedStorage
	:WaitForChild("Weapons")
	:WaitForChild("Chinelo")
	:WaitForChild("ChineloProjectile")

-- =====================
-- CONSTRUTOR
-- =====================
function Chinelo.new(player, limiter)

	local self = WeaponBase.new({
		Name = "Chinelo",
		Damage = 45,
		Range = 11,
		Cooldown = 2.2,
		ProjectileSpeed = 18
	}, player, limiter)

	self.Level = 1

	self.Scaling = {
		Damage = 1.15,
		Cooldown = 0.93
	}

	setmetatable(self, Chinelo)
	return self
end

-- =====================
-- EVOLUÇÃO
-- =====================
function Chinelo:OnUpgrade(level)

	self.Level = level

	self.BaseStats.Damage *= self.Scaling.Damage
	self.BaseStats.Cooldown *= self.Scaling.Cooldown

	print("🩴 Chinelo nível:", level)
end

-- =====================
-- PROJÉTIL VISUAL
-- =====================
local function spawnVisual(origin, direction, range, speed, sizeMultiplier, onDestroy)

	local model = ProjectileTemplate:Clone()
	model.Parent = workspace

	-- 🔥 GARANTE PRIMARY PART CORRETA
	if model:IsA("Model") then
		if not model.PrimaryPart then
			local base = model:FindFirstChild("R Base")
			if base then
				model.PrimaryPart = base
			else
				warn("ChineloProjectile sem PrimaryPart!")
				model:Destroy()
				return
			end
		end
	end

	local primary = model.PrimaryPart

	primary.Anchored = true
	primary.CanCollide = false
	primary.Massless = true

	-- 🔥 GARANTE SOLDAGEM
	for _, part in ipairs(model:GetDescendants()) do
		if part:IsA("BasePart") and part ~= model.PrimaryPart then
			local weld = Instance.new("WeldConstraint")
			weld.Part0 = model.PrimaryPart
			weld.Part1 = part
			weld.Parent = model.PrimaryPart

			part.Anchored = false
			part.CanCollide = false
			part.Massless = true
		end
	end

	model:PivotTo(CFrame.new(origin, origin + direction))

	local traveled = 0

	local conn
	conn = RunService.Heartbeat:Connect(function(dt)

		local step = speed * dt
		traveled += step

		if traveled >= range then
			conn:Disconnect()
			if onDestroy then onDestroy() end
			model:Destroy()
			return
		end

		local newPos =
			primary.Position + direction * step

		local newCF =
			CFrame.new(newPos, newPos + direction) *
			CFrame.Angles(
				math.rad(traveled * 35),
				math.rad(traveled * 15),
				0
			)

		model:PivotTo(newCF)
	end)

	Debris:AddItem(model, 6)

	task.delay(6, function()
		if onDestroy then onDestroy() end
	end)
end

-- =====================
-- ATAQUE
-- =====================
function Chinelo:Attack(root, attrs)

	local finalRange =
		self.BaseStats.Range
		* attrs.Range

	local enemies = self:GetEnemiesInRange(finalRange)
	if #enemies == 0 then return end

	self:StartCooldown(attrs)

	local multi = math.max(1, math.floor(attrs.MultiAttack or 1))

	for i = 1, math.min(multi, #enemies) do

		if not self:CanSpawnProjectile() then break end
		self:OnProjectileSpawn()

		local enemy = enemies[i]

		local finalDamage =
			self:ComputeFinalDamage(self.BaseStats.Damage, attrs)

		local dir =
			(enemy.Root.Position - root.Position).Unit

		local origin =
			root.Position + dir * 2 + Vector3.new(0, 1.6, 0)

		local finalSpeed =
			self.BaseStats.ProjectileSpeed
			* attrs.ProjectileSpeed

		spawnVisual(
			origin,
			dir,
			finalRange,
			finalSpeed,
			attrs.AttackSize,
			function()
				self:OnProjectileDestroy()
			end
		)

		EnemyDamageService:DamageEnemy(self.Player, enemy.Humanoid, finalDamage)

	end
end

return Chinelo
