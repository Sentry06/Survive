-- Knife.lua
-- Versão alinhada com sistema oficial de atributos

local RunService = game:GetService("RunService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Debris = game:GetService("Debris")
local TweenService = game:GetService("TweenService")

local EnemyDamageService = require(
	game.ServerScriptService.Combat.EnemyDamageService
)

local WeaponBase = require(script.Parent:WaitForChild("WeaponBase"))

local Knife = {}
Knife.__index = Knife
setmetatable(Knife,{ __index = WeaponBase })

-- =====================
-- TEMPLATES
-- =====================
local KnifeTemplate =
	ReplicatedStorage
	:WaitForChild("Weapons")
	:WaitForChild("Knife")
	:WaitForChild("KnifeProjectile")

-- =====================
-- CONSTRUTOR
-- =====================
function Knife.new(player, limiter)

	local baseStats = {
		Name = "Knife",

		-- 🔥 DOIS ATRIBUTOS BASE
		Damage = 14,
		Cooldown = 1.3,

		-- ⚙️ Atributos base fixos
		Range = 16,
		ProjectileSpeed = 22,
		AttackSize = 1,
		ProjectilePierce = 0,
		AreaDamage = 0
	}

	local self = WeaponBase.new(baseStats, player, limiter)

	self.Level = 1

	self.Scaling = {
		Damage = 1.10,   -- +10% por nível
		Cooldown = 0.94  -- -6% por nível
	}

	setmetatable(self, Knife)
	return self
end

-- =====================
-- EVOLUÇÃO
-- =====================
function Knife:OnUpgrade(level)

	self.Level = level

	self.BaseStats.Damage *= self.Scaling.Damage
	self.BaseStats.Cooldown *= self.Scaling.Cooldown

	print("🔪 Knife nível:", level)
end

-- =====================
-- AREA VISUAL
-- =====================
local function spawnAreaEffect(position, radius)

	if not AreaTemplate then return end

	local ring = AreaTemplate:Clone()
	ring.Parent = workspace

	ring.Size = Vector3.new(1, 0.2, 1)
	ring.CFrame =
		CFrame.new(position) *
		CFrame.Angles(math.rad(90), 0, 0)

	TweenService:Create(
		ring,
		TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
		{
			Size = Vector3.new(radius * 2, 0.2, radius * 2),
			Transparency = 1
		}
	):Play()

	Debris:AddItem(ring, 0.3)
end

-- =====================
-- PROJÉTIL VISUAL
-- =====================
local function spawnVisual(origin, direction, range, speed, sizeMultiplier, onDestroy)

	local knife = KnifeTemplate:Clone()
	knife.Parent = workspace

	local mesh = knife:FindFirstChildWhichIsA("FileMesh")
	if mesh then
		mesh.Scale = mesh.Scale * (1 + (sizeMultiplier - 1) * 3.5)
	end

	knife.CFrame = CFrame.new(origin, origin + direction)

	local traveled = 0
	local lastPos = origin

	local conn
	conn = RunService.Heartbeat:Connect(function(dt)

		local step = speed * dt
		traveled += step

		if traveled >= range then
			conn:Disconnect()

			if onDestroy then
				onDestroy()
			end

			knife:Destroy()
			return
		end

		local newPos = lastPos + direction * step

		knife.CFrame =
			CFrame.new(newPos, newPos + direction) *
			CFrame.Angles(0, math.rad(90), math.rad(traveled * 25))

		lastPos = newPos
	end)

	Debris:AddItem(knife, 5)
	if traveled >= range then
		conn:Disconnect()
		knife:Destroy()
		self:OnProjectileDestroy()
		return
	end
end

-- =====================
-- ATAQUE
-- =====================
function Knife:Attack(root, attrs)

	local finalRange =
		self.BaseStats.Range
		* attrs.Range

	local enemies = self:GetEnemiesInRange(finalRange)
	if #enemies == 0 then return end

	-- ⏱ Cooldown final
	self.Cooldown =
		self.BaseStats.Cooldown
		/ attrs.Cooldown

	local multi = math.max(1, math.floor(attrs.MultiAttack or 1))
	local pierce = math.max(0, math.floor(attrs.ProjectilePierce or 0))
	local areaPercent = math.max(0, attrs.AreaDamage or 0)
	local attackSize = attrs.AttackSize or 1

	local multi = math.max(1, math.floor(attrs.MultiAttack or 1))

	for i = 1, math.min(multi, #enemies) do

		local enemy = enemies[((i - 1) % #enemies) + 1]
		if not enemy or not enemy.Alive or not enemy.Root or not enemy.Humanoid then
			continue
		end

		local finalDamage =
			self.BaseStats.Damage
			* attrs.Damage

		-- 🎯 ALVO PRINCIPAL
		EnemyDamageService:DamageEnemy(
			self.Player,
			enemy.Humanoid,
			finalDamage
		)

		-- ➡️ PIERCE
		local hits = 0
		for _, other in ipairs(enemies) do
			if hits >= pierce then break end
			if other ~= enemy and other.Alive and other.Humanoid then
				other.Humanoid:TakeDamage(math.floor(finalDamage))
				hits += 1
			end
		end

		-- 💥 AREA DAMAGE
		if areaPercent > 0 then

			local radius = 4 * attackSize
			spawnAreaEffect(enemy.Root.Position, radius)

			for _, other in ipairs(enemies) do
				if other ~= enemy and other.Alive and other.Root and other.Humanoid then

					local dist =
						(other.Root.Position - enemy.Root.Position).Magnitude

					if dist <= radius then
						other.Humanoid:TakeDamage(
							math.floor(finalDamage * areaPercent)
						)
					end
				end
			end
		end

		-- ✨ VISUAL
		local dir =
			(enemy.Root.Position - root.Position).Unit

		local originOffset =
			math.clamp(finalRange * 0.15, 2, 8)

		local origin =
			root.Position
			+ dir * originOffset
			+ Vector3.new(0, 1.5, 0)

		spawnVisual(
			origin,
			dir,
			finalRange,
			self.BaseStats.ProjectileSpeed * attrs.ProjectileSpeed,
			attackSize,
			function()
				self:OnProjectileDestroy()
			end
		)
	end
end

return Knife
