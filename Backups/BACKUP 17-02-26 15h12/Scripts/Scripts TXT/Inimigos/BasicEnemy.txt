-- BasicEnemy.lua
-- Movimento físico do inimigo
-- PAUSA ABSOLUTA

local ReplicatedStorage = game:GetService("ReplicatedStorage")

local GameState = ReplicatedStorage:WaitForChild("GameState")
local GamePaused = GameState:WaitForChild("GamePaused")

local BasicEnemy = {}
BasicEnemy.__index = BasicEnemy

function BasicEnemy.new(model)
	local self = setmetatable({}, BasicEnemy)

	self.Model = model
	self.Humanoid = model:WaitForChild("Humanoid")
	self.Root = model:WaitForChild("HumanoidRootPart")

	self.Speed = 6
	self.Alive = true

	return self
end

function BasicEnemy:Update(targetPosition)
	if not self.Alive then
		return
	end

	-- ⏸️ PAUSA TOTAL
	if GamePaused.Value then
		-- 🌀 Se estiver sendo puxado pelo Gargantua, não controla movimento
		if self.Model:GetAttribute("BlackHolePull") then
			return
		end

		self.Root.AssemblyLinearVelocity = direction * self.Speed

		self.Root.Anchored = true
		return
	end

	-- 🔓 Retoma movimento
	if self.Root.Anchored then
		self.Root.Anchored = false
	end

	local direction = targetPosition - self.Root.Position
	direction = Vector3.new(direction.X, 0, direction.Z)

	if direction.Magnitude > 0 then
		direction = direction.Unit
	end

	-- 🌀 Se estiver sendo puxado pelo Gargantua, não controla movimento
	if self.Model:GetAttribute("BlackHolePull") then
		return
	end

	self.Root.AssemblyLinearVelocity = direction * self.Speed
end

function BasicEnemy:Destroy()
	self.Alive = false
	self.Root.AssemblyLinearVelocity = Vector3.zero
	self.Model:Destroy()
end

return BasicEnemy
