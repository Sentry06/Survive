-- XPService
local XPService = {}

local BASE_XP_TO_LEVEL = 20
local XP_GROWTH = 1.5

function XPService:GetXPRequired(level)
	return math.floor(BASE_XP_TO_LEVEL * (XP_GROWTH ^ (level - 1)))
end

function XPService:GetXPReward(baseXP, difficultyMultiplier, level)
	local levelBonus = 1 + (level * 0.05)
	return math.floor(baseXP * difficultyMultiplier * levelBonus)
end

return XPService
